Provider Interface — app/lib/ai/types.ts
// Every AI image provider implements this interface.
// Swap providers by changing AI_IMAGE_PROVIDER env var.

export interface GenerateImageRequest {
  prompt: string;          // Already amended by prompt engine
  count: number;           // Number of images (1-4)
  size: '1024x1024';      // Square for token design
  style?: 'natural' | 'vivid';
}

export interface GenerateImageResult {
  images: {
    url: string;            // Temporary URL from provider
    revisedPrompt?: string; // Provider's revised prompt (if any)
  }[];
  provider: string;          // 'openai' | 'stability' | etc.
  model: string;             // 'dall-e-3' | 'sd-xl' | etc.
  cost?: number;             // Estimated cost in cents (for logging)
}

export interface ImageGenerationProvider {
  generate(req: GenerateImageRequest): Promise<GenerateImageResult>;
  healthCheck(): Promise<boolean>;
}

import { OpenAIProvider } from './openai';

export function createImageProvider(env: Env): ImageGenerationProvider {
  const provider = env.AI_IMAGE_PROVIDER || 'openai';

  switch (provider) {
    case 'openai':
      return new OpenAIProvider(env.OPENAI_API_KEY);
    // case 'stability':
    //   return new StabilityProvider(env.STABILITY_API_KEY);
    default:
      throw new Error(`Unknown AI provider: ${provider}`);
  }
}
Prompt Engine — app/lib/ai/prompt-engine.ts

Purpose

Takes the customer's raw description and wraps it in token-specific design parameters. The customer never sees or controls these additions — they just describe what they want. 

const SYSTEM_PREFIX = `
Design a circular recovery/sobriety token coin.
The design must work as a physical engraved coin:
- Circular composition, fits within a circle border
- High contrast suitable for metal engraving
- Clean lines, no photographic elements
- Front face design (text/symbols centered)
- Metallic coin aesthetic
`;

const STYLE_SUFFIX = `
Style: detailed coin engraving illustration,
metallic surface, raised relief design,
professional commemorative coin aesthetic.
Top-down view of a single coin on dark background.
`;

export function buildTokenPrompt(
  customerPrompt: string,
  options?: {
    occasion?: string;
    material?: 'brass' | 'color';
  }
): string {
  const materialHint = options?.material === 'color'
    ? 'Colorful enamel coin design.'
    : 'Polished brass coin with silver engraving.';

  return [
    SYSTEM_PREFIX,
    materialHint,
    `Customer's design vision: ${customerPrompt}`,
    STYLE_SUFFIX,
  ].join('\n');
}
OpenAI Implementation — app/lib/ai/openai.ts

Day 1 Provider

Implements the interface using OpenAI's Images API (DALL-E 3). Generates one image per call (DALL-E 3 limitation), so 4 previews = 4 parallel calls. 

export class OpenAIProvider implements ImageGenerationProvider {
  private apiKey: string;

  constructor(apiKey: string) {
    this.apiKey = apiKey;
  }

  async generate(req: GenerateImageRequest) {
    // DALL-E 3 only does 1 image per call
    // Parallel calls for count > 1
    const promises = Array.from(
      { length: req.count },
      () => this.callOpenAI(req)
    );
    const results = await Promise.all(promises);

    return {
      images: results,
      provider: 'openai',
      model: 'dall-e-3',
      cost: req.count * 4, // ~$0.04 per 1024x1024
    };
  }

New Environment Variables
# AI Image Generation
AI_IMAGE_PROVIDER="openai"           # Which provider to use
OPENAI_API_KEY="sk-..."              # Production API key
OPENAI_API_KEY_PREVIEW="sk-..."      # Lower-limit key for preview deploys

# Rate Limiting
AI_MAX_GENERATIONS_PER_SESSION="7"  # 4 initial + 3 refinements
AI_MAX_GENERATIONS_PER_DAY="100"    # Global daily cap (cost protection)
Cost Protection: With DALL-E 3 at ~$0.04/image, a full "You Design" session (4 previews + 3 refinements = 7 images) costs ~$0.28. The per-session cap of 7 generations prevents abuse. The daily global cap prevents runaway costs during preview/testing. Both are configurable via env vars — no code change needed to adjust.
